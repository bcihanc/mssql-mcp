> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `@bcihanc/mysql-mcp` — a read-only MySQL MCP server — by porting the existing `@bcihanc/mssql-mcp` template to the `mysql2` driver and the MySQL `information_schema` dialect.

**Architecture:** A new, standalone npm package. We copy the `mssql-mcp` source tree as the starting point, then transform it file-by-file: swap the driver (`mssql` → `mysql2/promise`), convert SQL from `sys.*`/DMV to `information_schema`, change identifier quoting from `[brackets]` to `` `backticks` ``, collapse the MSSQL three-level `database.schema.table` model to MySQL's two-level `database.table`, and add MySQL-specific read-only enforcement (session `READ ONLY` + `multipleStatements:false` + file-exfiltration blacklist). All DB-independent infrastructure (resilient pool wrapper, LRU+TTL caches, CSV utilities, STDIO+HTTP transport, esbuild bundler) is preserved with only naming changes.

**Tech Stack:** TypeScript (ESM), Node 18+, `mysql2`, `@modelcontextprotocol/sdk`, `zod` v4, `hono` + `@hono/mcp`, `commander`, `consola`, `esbuild`.

## Global Constraints

- **Read-only is non-negotiable.** Every write path (INSERT/UPDATE/DELETE/DDL/DCL/file-exfil) must be blocked. Enforcement is layered: query validation (`isReadOnlyQuery`) + driver (`multipleStatements: false`) + session (`SET SESSION TRANSACTION READ ONLY`).
- **MySQL 5.7 + 8.x compatibility.** Use only `information_schema` / `performance_schema` columns present in BOTH 5.7 and 8.0. Do not use 8.0-only catalog columns.
- **Schema == database.** There is no intermediate schema layer. Tool params use `database_name` (never `schema_name`). Identifiers are at most 2 parts (`database.table`).
- **Identifier safety.** Interpolated identifiers go through `validateObjectName` / `validateDatabaseName` only; they are backtick-quoted. String literals interpolated into SQL are escaped by doubling `'`.
- **Node >= 18.0.0.** Package is ESM (`"type": "module"`).
- **Package identity:** name `@bcihanc/mysql-mcp`, binary `mysql-mcp` → `dist/main.mjs`, MCP server name `mysql-mcp-server`, resource URI scheme `mysql://`.
- **Implementation target directory:** `/Users/cihan/mcp/mysql-mcp` (a NEW sibling directory and NEW git repo). The template lives at `/Users/cihan/mcp/mssql-mcp`.
- **Tool count: 17.** Drop `list_schemas` and `list_linked_servers` (no MySQL equivalent).
- **Tests run without a live database.** Validation/identifier tests are pure-logic. Tool tests use a fake pool that captures the generated SQL string.

```ts
import { validateObjectName } from '../utils/identifier.js';

export interface MysqlConfig {
	host: string;
	database: string;
	user?: string;
	password?: string;
	port: number;
	ssl: boolean;
}

const logger = consola.withTag('mysql-config');

/**
 * Get database configuration from MYSQL_* environment variables.
 */
export function getMysqlConfig(): MysqlConfig {
	const host = process.env.MYSQL_HOST || 'localhost';
	const config: MysqlConfig = {
		host,
		user: process.env.MYSQL_USER,
		password: process.env.MYSQL_PASSWORD,
		database: process.env.MYSQL_DATABASE || '',
		port: 3306,
		ssl: process.env.MYSQL_SSL?.toLowerCase() === 'true',
	};

	const port = process.env.MYSQL_PORT;
	if (port) {
		const parsed = parseInt(port, 10);
		if (!Number.isNaN(parsed)) {
			config.port = parsed;
		} else if (consola.level >= 0) {
			logger.warn(`Invalid MYSQL_PORT value: ${port}. Using default port 3306.`);
		}
	}

	if (!config.user || !config.password || !config.database) {
		if (consola.level >= 0) {
			logger.error('Missing required configuration: MYSQL_USER, MYSQL_PASSWORD, and MYSQL_DATABASE are required');
		}
		throw new Error('Missing required database configuration');
	}

	if (consola.level >= 0) {
		logger.info(`Database config: ${config.host}:${config.port}/${config.database} as ${config.user} (READ-ONLY mode)`);
	}

	return config;
}

/**
 * Check if a SQL query is a read-only operation with layered security validation.
 * Protects against Unicode normalization, URL-encoding, comment, and hex-encoding bypasses.
 */
export function isReadOnlyQuery(query: string): boolean {
	let decodedQuery = query;
	try {
		decodedQuery = decodeURIComponent(decodedQuery);
	} catch {
		// already-decoded or invalid encoding — use original
	}

	decodedQuery = decodedQuery.normalize('NFKC');

	const cleanQuery = decodedQuery
		.replace(/--.*$/gm, '')
		.replace(/\/\*[\s\S]*?\*\//g, '')
		.replace(/\\/g, '')
		.trim()
		.toUpperCase();

	if (!cleanQuery) {
		if (consola.level >= 0) logger.warn('Empty query after sanitization');
		return false;
	}

	// Dangerous patterns: DDL, DML, DCL, exec, admin, file-exfiltration, multi-statement, hex.
	const dangerousPattern =
		/\b(DROP|TRUNCATE|ALTER|CREATE|RENAME|INSERT|UPDATE|DELETE|REPLACE|MERGE|GRANT|REVOKE|CALL|HANDLER|LOAD_FILE|LOCK|UNLOCK|FLUSH|RESET|SHUTDOWN|KILL|UNION|INTO|OUTFILE|DUMPFILE)\b|\bLOAD\s+DATA\b|;\s*(DROP|TRUNCATE|ALTER|INSERT|UPDATE|DELETE|CREATE|REPLACE|CALL)|0X[0-9A-F]+/i;

	if (dangerousPattern.test(cleanQuery)) {
		if (consola.level >= 0) logger.warn('Dangerous pattern detected in query');
		return false;
	}

	if (/0X[0-9A-F]{8,}/i.test(cleanQuery)) {
		if (consola.level >= 0) logger.warn('Hex-encoded content detected - potential bypass attempt');
		return false;
	}

	const readOnlyPattern = /^(SELECT|WITH|SHOW|DESCRIBE|EXPLAIN|DESC)\b/;
	if (!readOnlyPattern.test(cleanQuery)) {
		if (consola.level >= 0) {
			logger.warn(`Query does not start with allowed read-only operation: ${cleanQuery.substring(0, 50)}`);
		}
		return false;
	}

	return true;
}

/**
 * @deprecated Use `validateObjectName` from `../utils/identifier` instead.
 */
export function validateTableName(tableName: string): string {
	return validateObjectName(tableName);
}
```

```ts
/**
 * MySQL identifier validation and backtick-quoting utilities.
 *
 * Supports 1-part (object) or 2-part (database.object) names. MySQL has no
 * intermediate schema layer — schema == database. All parts must contain only
 * alphanumeric characters, underscore, or `$`. Identifiers are backtick-quoted.
 *
 * SECURITY: This is the ONLY safe way to interpolate identifiers into SQL.
 */

const PART_REGEX = /^[a-zA-Z0-9_$]+$/;
const FULL_NAME_REGEX = /^[a-zA-Z0-9_$]+(\.[a-zA-Z0-9_$]+)?$/;

export interface ObjectNameParts {
	database?: string;
	object: string;
}

export function parseObjectName(name: string): ObjectNameParts {
	if (!name || typeof name !== 'string') {
		throw new Error('Object name must be a non-empty string');
	}
	if (!FULL_NAME_REGEX.test(name)) {
		throw new Error(
			`Invalid object name: "${name}". Allowed: alphanumeric, underscore and $ in each part, separated by a single dot (1-2 parts).`,
		);
	}
	const parts = name.split('.');
	if (parts.length === 1) return { object: parts[0] };
	return { database: parts[0], object: parts[1] };
}

export function validateObjectName(name: string): string {
	const parts = parseObjectName(name);
	const out: string[] = [];
	if (parts.database) out.push(`\`${parts.database}\``);
	out.push(`\`${parts.object}\``);
	return out.join('.');
}

export function validateDatabaseName(name: string): string {
	if (!name || typeof name !== 'string') {
		throw new Error('Database name must be a non-empty string');
	}
	if (!PART_REGEX.test(name)) {
		throw new Error(`Invalid database name: "${name}". Only alphanumeric characters, underscore and $ are allowed.`);
	}
	return `\`${name}\``;
}

export function buildCacheKeyPrefix(dbContext?: string): string {
	if (!dbContext) return '_default_::';
	if (!PART_REGEX.test(dbContext)) {
		throw new Error(`Invalid database context for cache key: "${dbContext}"`);
	}
	return `${dbContext.toLowerCase()}::`;
}
```

```ts
import { getFileLogger } from '../utils/fileLogger';
import type { MysqlConfig } from './config';

const logger = consola.withTag('mysql-connection');

export interface ConnectionPool {
	query<T = any>(sqlQuery: string): Promise<T[]>;
	close(): Promise<void>;
}

function buildPoolOptions(config: MysqlConfig): mysql.PoolOptions {
	if (!config.user || !config.password) {
		throw new Error('Username and password are required for MySQL authentication');
	}
	return {
		host: config.host,

		user: config.user,
		password: config.password,
		database: config.database,
		waitForConnections: true,
		connectionLimit: 10,
		queueLimit: 0,
		// SECURITY: forbid stacked statements (SELECT 1; DROP TABLE ...) at the driver level
		multipleStatements: false,
		ssl: config.ssl ? {} : undefined,
		dateStrings: true,
	};

/**
 * Classify and handle query execution errors.
 * Schema/syntax errors are re-thrown verbatim; only genuine write-block errors are wrapped.
 */
function handleQueryError(error: unknown, sqlQuery: string): never {
	const fileLogger = getFileLogger();
	if (consola.level >= 0) logger.error('Query execution failed:', error);

	const errorMessage = error instanceof Error ? error.message : 'Unknown error';
	const lower = errorMessage.toLowerCase();

	fileLogger.error('Query execution failed', { errorMessage, query: sqlQuery.substring(0, 200) });

	// FIRST: schema/syntax errors (MySQL phrasings) — re-throw as-is.
	if (
		lower.includes('unknown column')
		|| lower.includes("doesn't exist")
		|| lower.includes('unknown table')
		|| lower.includes('unknown database')
		|| lower.includes('you have an error in your sql syntax')
		|| lower.includes('ambiguous')
	) {
		throw error;
	}

	// SECOND: read-only violation phrasings.
	if (
		lower.includes('read only')
		|| lower.includes('read-only')
		|| lower.includes('--read-only')
		|| lower.includes('command denied')
	) {
		const writeError = `READ-ONLY mode violation: Write operation detected and blocked. ${errorMessage}`;
		fileLogger.error(writeError);
		throw new Error(writeError);
	}

	throw error;
}

function isConnectionError(error: unknown): boolean {
	const code = (error as { code?: string })?.code?.toUpperCase() || '';
	const msg = (error instanceof Error ? error.message : String(error)).toLowerCase();
	return (
		code === 'PROTOCOL_CONNECTION_LOST'
		|| code === 'ECONNRESET'
		|| code === 'ECONNREFUSED'
		|| code === 'ETIMEDOUT'
		|| code === 'EPIPE'
		|| code === 'ENOTFOUND'
		|| msg.includes('connection lost')
		|| msg.includes('connection is closed')
		|| msg.includes('socket hang up')
	);
}

export class ResilientConnectionPool implements ConnectionPool {
	private pool: mysql.Pool | null = null;
	private poolOptions: mysql.PoolOptions;
	private localConfig: MysqlConfig;
	private connected = false;
	private connectingPromise: Promise<boolean> | null = null;
	private retryTimer: ReturnType<typeof setTimeout> | null = null;
	private retryDelay = 1000;
	private readonly maxRetryDelay = 60000;
	private stopped = false;

	get isConnected(): boolean {
		return this.connected;
	}

	constructor(config: MysqlConfig) {
		this.localConfig = config;
		this.poolOptions = buildPoolOptions(config);
		const fileLogger = getFileLogger();
		if (consola.level >= 0) logger.info('Connection configured for READ-ONLY access mode');
		fileLogger.info('ResilientConnectionPool created (READ-ONLY mode)');
	}

	async ensureConnected(): Promise<boolean> {
		if (this.connected) return true;
		if (this.stopped) return false;
		if (this.connectingPromise) return this.connectingPromise;

		this.connectingPromise = this.doConnect();
		try {
			const result = await this.connectingPromise;
			if (!result && !this.stopped) this.scheduleBackgroundRetry();
			return result;
		} finally {
			this.connectingPromise = null;
		}
	}

	private async doConnect(): Promise<boolean> {
		const fileLogger = getFileLogger();
		fileLogger.info('Attempting to connect to database...');

		if (this.pool) {
			try { await this.pool.end(); } catch { /* ignore */ }
			this.pool = null;
		}

		try {
			const newPool = mysql.createPool(this.poolOptions);

			// SECURITY: every new physical connection starts in a read-only transaction context.
			// In autocommit mode each statement is its own transaction, so this applies to all reads.
			newPool.on('connection', (conn: any) => {
				conn.query('SET SESSION TRANSACTION READ ONLY', (err: unknown) => {
					if (err) fileLogger.error('Failed to set session READ ONLY', { error: String(err) });
				});
			});

			// Verify connectivity with a trivial read.
			await newPool.query('SELECT 1');
			this.pool = newPool;
			this.connected = true;
			this.retryDelay = 1000;
			this.clearRetryTimer();
			fileLogger.info('Connection pool connected successfully (READ-ONLY mode)');
			return true;
		} catch (error) {
			fileLogger.error('Database connection attempt failed', {
				error: error instanceof Error ? error.message : String(error),
			});
			if (this.pool) { try { await this.pool.end(); } catch { /* ignore */ } }
			this.pool = null;
			this.connected = false;
			return false;
		}
	}

	private scheduleBackgroundRetry(): void {
		if (this.stopped || this.connected || this.retryTimer) return;
		const fileLogger = getFileLogger();
		fileLogger.info(`Scheduling background reconnection attempt in ${this.retryDelay}ms`);
		this.retryTimer = setTimeout(async () => {
			this.retryTimer = null;
			if (this.stopped || this.connected) return;
			const success = await this.ensureConnected();
			if (!success && !this.stopped) {
				this.retryDelay = Math.min(this.retryDelay * 2, this.maxRetryDelay);
				this.scheduleBackgroundRetry();
			}
		}, this.retryDelay);
	}

	private clearRetryTimer(): void {
		if (this.retryTimer) { clearTimeout(this.retryTimer); this.retryTimer = null; }
	}

	async query<T = any>(sqlQuery: string): Promise<T[]> {
		const fileLogger = getFileLogger();

		if (!this.connected) {
			const success = await this.ensureConnected();
			if (!success) {
				throw new Error('Database is currently unavailable. The server will automatically reconnect when the database becomes available.');
			}
		}

		const pool = this.pool;
		if (!pool) {
			throw new Error('Database is currently unavailable. The server will automatically reconnect when the database becomes available.');
		}

		fileLogger.debug('Executing query', { query: sqlQuery.substring(0, 200) });
		try {
			const [rows] = await pool.query(sqlQuery);
			fileLogger.debug('Query executed successfully', {
				rowCount: Array.isArray(rows) ? rows.length : 0,
			});
			return (Array.isArray(rows) ? rows : []) as T[];
		} catch (error) {
			if (isConnectionError(error)) {
				fileLogger.error('Connection lost during query execution, starting background retry');
				this.connected = false;
				if (!this.stopped) {
					this.retryDelay = 1000;
					this.scheduleBackgroundRetry();
				}
				throw new Error('Database connection was lost during query execution. The server will automatically reconnect when the database becomes available.');
			}
			return handleQueryError(error, sqlQuery);
		}
	}

	async close(): Promise<void> {
		this.stopped = true;
		this.clearRetryTimer();
		if (this.pool) {
			try { await this.pool.end(); } catch { /* ignore */ }
			this.pool = null;
		}
		this.connected = false;
		if (consola.level >= 0) logger.info('Connection pool closed');
	}
}

export function createResilientConnectionPool(config: MysqlConfig): ResilientConnectionPool {
	const fileLogger = getFileLogger();
	fileLogger.info('createResilientConnectionPool() called', {
		host: config.host, database: config.database, port: config.port, ssl: config.ssl,
	});
	return new ResilientConnectionPool(config);
}
```

```ts
async handleRowCount(args: any, pool: ConnectionPool): Promise<{ content: TextContent[] }> {
	try {
		const v = RowCountInputSchema.parse(args);
		const table = resolveTable(v.table_name, v.database_name);
		const cacheKey = `${table.dbCacheKey}${table.tableName}:${v.exact ? 'exact' : 'fast'}`;

		const cached = getFromCache(rowCountCache, cacheKey, ROW_COUNT_CACHE_TTL_MS);
		if (cached !== null) return cachedResponse(cached);

		const tableLit = `'${escapeLiteral(table.tableName)}'`;
		let rowCount: number | null = null;
		let source = 'unknown';

		// Tier 1: fast metadata estimate from information_schema.TABLES (approximate for InnoDB).
		if (!v.exact) {
			try {
				const r = await pool.query<{ row_count: number | null }>(
					`SELECT TABLE_ROWS AS row_count FROM information_schema.TABLES WHERE TABLE_SCHEMA = ${table.schemaExpr} AND TABLE_NAME = ${tableLit}`,
				);
				const n = r[0]?.row_count;
				if (n != null && Number(n) > 0) {
					rowCount = Number(n);
					source = 'information_schema.TABLES.TABLE_ROWS (estimate; approximate for InnoDB)';
				}
			} catch {
				// fall through to exact
			}
		}

		// Tier 2: exact COUNT(*).
		if (rowCount == null) {
			try {
				const r = await pool.query<{ cnt: number }>(`SELECT COUNT(*) AS cnt FROM ${table.bracketedFqn}`);
				rowCount = Number(r[0]?.cnt ?? 0);
				source = v.exact ? 'COUNT(*) (exact)' : 'COUNT(*) (fallback after estimate returned 0/unavailable)';
			} catch (e) {
				if (isPermissionError(e)) {
					return plainResponse(`🔒 ${table.displayName} — cannot count rows. Connection user lacks SELECT permission on this table.`);
				}
				throw e;
			}
		}

		const text = `Row count for ${table.displayName}: ${rowCount}\nSource: ${source}`;
		setInCache(rowCountCache, cacheKey, text, ROW_COUNT_CACHE_MAX_SIZE, 'get_table_row_count');
		return plainResponse(text);
	} catch (error) {
		if (consola.level >= 0) logger.error('get_table_row_count error:', error);
		return errorResponse('Error getting row count', error);
	}
}
```
